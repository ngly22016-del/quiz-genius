
import React, { useState, useRef, useMemo, useEffect } from 'react';
import { Question, Difficulty, QuestionType, Year, Subject, Topic, QuizSet, RegisteredUser, UserRole, QuizAttempt } from '../types';
import { extractQuestionsFromText } from '../services/geminiService';
import MathText from './MathText';

interface AdminPanelProps {
  questions: Question[];
  years: Year[];
  subjects: Subject[];
  topics: Topic[];
  subTopics: any[];
  quizSets: QuizSet[];
  registeredUsers: RegisteredUser[];
  attempts: QuizAttempt[];
  onAddQuizSet: (name: string, description: string) => void;
  onUpdateQuizSet: (set: QuizSet) => void;
  onDeleteQuizSet: (id: string) => void;
  onAddYear: (name: string) => void;
  onUpdateYear: (id: string, name: string) => void;
  onDeleteYear: (id: string) => void;
  onAddSubject: (name: string) => void;
  onUpdateSubject: (id: string, name: string) => void;
  onDeleteSubject: (id: string) => void;
  onAddTopic: (name: string) => void;
  onUpdateTopic: (id: string, newName: string) => void;
  onDeleteTopic: (id: string) => void;
  onAddSubTopic: (name: string, topicId: string) => void;
  onDeleteSubTopic: (id: string) => void;
  onAddQuestions: (newQuestions: Question[]) => void;
  onUpdateQuestion: (question: Question) => void;
  onDeleteQuestion: (id: string) => void;
  onDeleteTopicQuestions: (topic: string) => void;
  onToggleUserStatus: (username: string) => void;
  onAddStudent: (user: RegisteredUser) => void;
  onDeleteStudent: (username: string) => void;
  onResetPassword: (username: string, newPass: string) => void;
  onDeleteAttempt: (id: string) => void;
  onRecordAttempt: (att: QuizAttempt) => void;
}

declare const pdfjsLib: any;

type SortField = 'year' | 'subject' | 'topic' | 'type' | 'difficulty' | 'none';
type SortOrder = 'asc' | 'desc';

type UndoType = 'question' | 'student' | 'attempt' | 'quizSet';

const AdminPanel: React.FC<AdminPanelProps> = ({ 
  questions, 
  years,
  subjects,
  topics,
  quizSets,
  registeredUsers,
  attempts,
  onAddQuizSet,
  onUpdateQuizSet,
  onDeleteQuizSet,
  onAddYear,
  onAddSubject,
  onAddTopic,
  onAddQuestions, 
  onUpdateQuestion,
  onDeleteQuestion,
  onToggleUserStatus,
  onAddStudent,
  onDeleteStudent,
  onResetPassword,
  onDeleteAttempt,
  onRecordAttempt
}) => {
  const [activeTab, setActiveTab] = useState<'generate' | 'manual' | 'manage' | 'bank' | 'sets' | 'students'>('generate');
  const [searchTerm, setSearchTerm] = useState('');
  
  // Sorting & Filtering
  const [sortField, setSortField] = useState<SortField>('none');
  const [sortOrder, setSortOrder] = useState<SortOrder>('asc');
  const [filterYear, setFilterYear] = useState<string>('All');
  const [filterSubject, setFilterSubject] = useState<string>('All');
  const [filterTopic, setFilterTopic] = useState<string>('All');
  const [filterDifficulty, setFilterDifficulty] = useState<string>('All');
  const [filterType, setFilterType] = useState<string>('All');

  // Question State
  const [expandedQuestionId, setExpandedQuestionId] = useState<string | null>(null);
  const [editingQuestion, setEditingQuestion] = useState<Question | null>(null);

  // Generation Progress
  const [isProcessing, setIsProcessing] = useState(false);
  const [progressValue, setProgressValue] = useState(0);
  const [selectedFile, setSelectedFile] = useState<File | null>(null);
  const fileInputRef = useRef<HTMLInputElement>(null);
  const manualDraftInputRef = useRef<HTMLInputElement>(null);

  const [selectedYear, setSelectedYear] = useState(years[0]?.name || '');
  const [selectedSubject, setSelectedSubject] = useState(subjects[0]?.name || '');
  const [selectedTopicName, setSelectedTopicName] = useState(topics[0]?.name || '');
  const [genDifficulty, setGenDifficulty] = useState<Difficulty>('Medium');
  const [genType, setGenType] = useState<QuestionType>('MCQ');

  // Registration Form State
  const [regForm, setRegForm] = useState({ username: '', password: '' });

  // Manual Entry Form
  const [manualForm, setManualForm] = useState<{
    question: string;
    options: string[];
    optionImages: (string | undefined)[];
    correctAnswer: string;
    explanation: string;
    difficulty: Difficulty;
    type: QuestionType;
    year: string;
    subject: string;
    topic: string;
    image?: string;
  }>({
    question: '',
    options: ['', '', '', ''],
    optionImages: [undefined, undefined, undefined, undefined],
    correctAnswer: '',
    explanation: '',
    difficulty: 'Medium',
    type: 'MCQ',
    year: years[0]?.name || '',
    subject: subjects[0]?.name || '',
    topic: topics[0]?.name || '',
    image: undefined
  });

  const [showManualPreview, setShowManualPreview] = useState(false);
  const [showEditPreview, setShowEditPreview] = useState(false);

  // Quiz Set Editing
  const [editingQuizSet, setEditingQuizSet] = useState<QuizSet | null>(null);
  const [setEditorTab, setSetEditorTab] = useState<'content' | 'enrollment'>('content');
  const [newSetName, setNewSetName] = useState('');

  // Undo System State
  const [undoToast, setUndoToast] = useState<{ show: boolean, type: UndoType, item: any | null }>({ show: false, type: 'question', item: null });
  const undoTimerRef = useRef<any>(null);

  // Student Views
  const [viewingStudentRecords, setViewingStudentRecords] = useState<string | null>(null);
  const [viewingAttemptDetailId, setViewingAttemptDetailId] = useState<string | null>(null);
  const [visiblePasswords, setVisiblePasswords] = useState<Set<string>>(new Set());
  const [editingPasswordFor, setEditingPasswordFor] = useState<{ username: string, newPass: string } | null>(null);

  // Confirmation Modals
  const [confirmDelete, setConfirmDelete] = useState<{ type: UndoType, id: string, name?: string } | null>(null);

  const difficultyOrder: Record<Difficulty, number> = { 'Easy': 1, 'Medium': 2, 'Hard': 3 };
  const typeOrder: Record<QuestionType, number> = { 'MCQ': 1, 'STRUCTURE': 2, 'ESSAY': 3 };

  const students = registeredUsers.filter(u => u.role === UserRole.STUDENT);

  const filteredQuestions = useMemo(() => {
    let result = questions.filter(q => {
      const matchesSearch = q.question.toLowerCase().includes(searchTerm.toLowerCase()) || q.topic.toLowerCase().includes(searchTerm.toLowerCase());
      const matchesYear = filterYear === 'All' || q.year === filterYear;
      const matchesSubject = filterSubject === 'All' || q.subject === filterSubject;
      const matchesTopic = filterTopic === 'All' || q.topic === filterTopic;
      const matchesDifficulty = filterDifficulty === 'All' || q.difficulty === filterDifficulty;
      const matchesType = filterType === 'All' || q.type === filterType;
      return matchesSearch && matchesYear && matchesSubject && matchesTopic && matchesDifficulty && matchesType;
    });

    if (sortField !== 'none') {
      result = [...result].sort((a, b) => {
        let valA: any = a[sortField as keyof Question];
        let valB: any = b[sortField as keyof Question];
        if (sortField === 'difficulty') { valA = difficultyOrder[a.difficulty]; valB = difficultyOrder[b.difficulty]; }
        if (sortField === 'type') { valA = typeOrder[a.type]; valB = typeOrder[b.type]; }
        if (valA < valB) return sortOrder === 'asc' ? -1 : 1;
        if (valA > valB) return sortOrder === 'asc' ? 1 : -1;
        return 0;
      });
    }
    return result;
  }, [questions, searchTerm, filterYear, filterSubject, filterTopic, filterDifficulty, filterType, sortField, sortOrder]);

  const handleProcessUpload = async (isManualDraft: boolean = false) => {
    const fileToUse = isManualDraft ? manualDraftInputRef.current?.files?.[0] : selectedFile;
    const yearToUse = isManualDraft ? manualForm.year : selectedYear;
    const subjectToUse = isManualDraft ? manualForm.subject : selectedSubject;
    const topicToUse = isManualDraft ? manualForm.topic : selectedTopicName;

    if (!fileToUse || !topicToUse || !yearToUse || !subjectToUse) {
      return alert("Please ensure Year, Subject, Topic and a file are selected.");
    }

    setIsProcessing(true); setProgressValue(10);
    try {
      const arrayBuffer = await fileToUse.arrayBuffer();
      let extractedText = '';
      let pageImages: string[] = [];

      if (fileToUse.name.toLowerCase().endsWith('.pdf')) {
        pdfjsLib.GlobalWorkerOptions.workerSrc = 'https://cdnjs.cloudflare.com/ajax/libs/pdf.js/3.11.174/pdf.worker.min.js';
        const pdf = await pdfjsLib.getDocument({ data: arrayBuffer }).promise;
        const canvas = document.createElement('canvas');
        const context = canvas.getContext('2d');
        const pagesToProcess = isManualDraft ? Math.min(pdf.numPages, 1) : pdf.numPages;

        for (let i = 1; i <= pagesToProcess; i++) {
          const page = await pdf.getPage(i);
          const content = await page.getTextContent();
          extractedText += content.items.map((it: any) => it.str).join(' ') + ' ';
          const viewport = page.getViewport({ scale: 1.5 });
          canvas.height = viewport.height;
          canvas.width = viewport.width;
          if (context) {
            await page.render({ canvasContext: context, viewport: viewport }).promise;
            pageImages.push(canvas.toDataURL('image/jpeg', 0.8));
          }
          setProgressValue(Math.floor((i / pagesToProcess) * 70));
        }
      } else {
        throw new Error("High-fidelity extraction is currently optimized for PDF documents.");
      }

      const extracted = await extractQuestionsFromText(
        extractedText, 
        yearToUse, 
        subjectToUse, 
        topicToUse, 
        '', 
        '', 
        isManualDraft ? manualForm.difficulty : genDifficulty, 
        isManualDraft ? manualForm.type : genType, 
        isManualDraft ? "EXTRACT ONLY THE FIRST QUESTION FOUND." : "", 
        pageImages
      );

      if (isManualDraft && extracted.length > 0) {
        const q = extracted[0];
        setManualForm(prev => ({
          ...prev,
          question: q.question,
          options: q.options.length > 0 ? q.options : prev.options,
          optionImages: q.optionImages || [undefined, undefined, undefined, undefined],
          correctAnswer: q.correctAnswer,
          explanation: q.explanation,
          difficulty: q.difficulty,
          type: q.type
        }));
        alert("Draft populated from file!");
      } else {
        onAddQuestions(extracted);
        setProgressValue(100);
        setTimeout(() => setActiveTab('bank'), 500);
      }
    } catch (err) { 
      alert(`Error: ${err instanceof Error ? err.message : "Invalid document processing."}`); 
    } finally { 
      setIsProcessing(false); setProgressValue(0); 
    }
  };

  const handleEditSubmit = (e: React.FormEvent) => {
    e.preventDefault();
    if (editingQuestion) {
      onUpdateQuestion(editingQuestion);
      setEditingQuestion(null);
    }
  };

  const handleManualSubmit = (e: React.FormEvent) => {
    e.preventDefault();
    if (!manualForm.question || !manualForm.correctAnswer) return alert("Fill required fields.");
    const newQ: Question = {
      id: `q-man-${Date.now()}`,
      ...manualForm,
      subTopic: '',
      section: ''
    };
    onUpdateQuestion(newQ);
    alert("Question added to bank.");
    setManualForm({
      question: '',
      options: ['', '', '', ''],
      optionImages: [undefined, undefined, undefined, undefined],
      correctAnswer: '',
      explanation: '',
      difficulty: 'Medium',
      type: 'MCQ',
      year: years[0]?.name || '',
      subject: subjects[0]?.name || '',
      topic: topics[0]?.name || '',
      image: undefined
    });
  };

  const handleQuestionImageUpload = (isEdit: boolean = false) => {
    const input = document.createElement('input');
    input.type = 'file';
    input.accept = 'image/*';
    input.onchange = async (e: any) => {
      const file = e.target.files?.[0];
      if (file) {
        const reader = new FileReader();
        reader.onload = (re: any) => {
          const base64 = re.target.result;
          if (isEdit && editingQuestion) {
            setEditingQuestion({ ...editingQuestion, image: base64 as string });
          } else if (!isEdit) {
            setManualForm({ ...manualForm, image: base64 as string });
          }
        };
        reader.readAsDataURL(file);
      }
    };
    input.click();
  };

  // Fix: Line 321 error was here. Removed reference to undefined 'base64'.
  const removeQuestionImage = (isEdit: boolean = false) => {
    if (isEdit && editingQuestion) {
      setEditingQuestion({ ...editingQuestion, image: undefined });
    } else if (!isEdit) {
      setManualForm({ ...manualForm, image: undefined });
    }
  };

  const handleOptionImageUpload = (index: number, isEdit: boolean = false) => {
    const input = document.createElement('input');
    input.type = 'file';
    input.accept = 'image/*';
    input.onchange = async (e: any) => {
      const file = e.target.files?.[0];
      if (file) {
        const reader = new FileReader();
        reader.onload = (re: any) => {
          const base64 = re.target.result;
          if (isEdit && editingQuestion) {
            const newOptionImages = [...(editingQuestion.optionImages || [undefined, undefined, undefined, undefined])];
            newOptionImages[index] = base64 as string;
            setEditingQuestion({ ...editingQuestion, optionImages: newOptionImages });
          } else if (!isEdit) {
            const newOptionImages = [...manualForm.optionImages];
            newOptionImages[index] = base64 as string;
            setManualForm({ ...manualForm, optionImages: newOptionImages });
          }
        };
        reader.readAsDataURL(file);
      }
    };
    input.click();
  };

  const removeOptionImage = (index: number, isEdit: boolean = false) => {
    if (isEdit && editingQuestion) {
      const newOptionImages = [...(editingQuestion.optionImages || [undefined, undefined, undefined, undefined])];
      newOptionImages[index] = undefined;
      setEditingQuestion({ ...editingQuestion, optionImages: newOptionImages });
    } else if (!isEdit) {
      const newOptionImages = [...manualForm.optionImages];
      newOptionImages[index] = undefined;
      setManualForm({ ...manualForm, optionImages: newOptionImages });
    }
  };

  const handleRegisterStudent = (e: React.FormEvent) => {
    e.preventDefault();
    if (!regForm.username || !regForm.password) return alert("Username and password are required.");
    const existing = registeredUsers.find(u => u.username.toLowerCase() === regForm.username.toLowerCase());
    if (existing) return alert("Username already taken.");

    onAddStudent({
      username: regForm.username,
      password: regForm.password,
      role: UserRole.STUDENT,
      isActive: true
    });
    setRegForm({ username: '', password: '' });
    alert("Student registered successfully.");
  };

  const handleDeleteWithUndo = (type: UndoType, id: string) => {
    if (undoTimerRef.current) clearTimeout(undoTimerRef.current);
    
    let itemToUndo: any = null;
    if (type === 'question') {
      itemToUndo = questions.find(q => q.id === id);
      onDeleteQuestion(id);
    } else if (type === 'student') {
      itemToUndo = registeredUsers.find(u => u.username === id);
      onDeleteStudent(id);
    } else if (type === 'attempt') {
      itemToUndo = attempts.find(a => a.id === id);
      onDeleteAttempt(id);
    } else if (type === 'quizSet') {
      itemToUndo = quizSets.find(s => s.id === id);
      onDeleteQuizSet(id);
    }

    setUndoToast({ show: true, type, item: itemToUndo });
    undoTimerRef.current = setTimeout(() => {
      setUndoToast({ show: false, type: 'question', item: null });
    }, 6000);
  };

  const handleUndo = () => {
    if (undoToast.item) {
      if (undoToast.type === 'question') onUpdateQuestion(undoToast.item);
      else if (undoToast.type === 'student') onAddStudent(undoToast.item);
      else if (undoToast.type === 'attempt') onRecordAttempt(undoToast.item);
      else if (undoToast.type === 'quizSet') onUpdateQuizSet(undoToast.item);
      
      setUndoToast({ show: false, type: 'question', item: null });
      if (undoTimerRef.current) clearTimeout(undoTimerRef.current);
    }
  };

  const handleConfirmAction = () => {
    if (confirmDelete) {
      handleDeleteWithUndo(confirmDelete.type, confirmDelete.id);
      setConfirmDelete(null);
    }
  };

  const handlePasswordResetSubmit = (e: React.FormEvent) => {
    e.preventDefault();
    if (editingPasswordFor && editingPasswordFor.newPass.trim()) {
      onResetPassword(editingPasswordFor.username, editingPasswordFor.newPass.trim());
      alert(`Access key for ${editingPasswordFor.username} has been rotated.`);
      setEditingPasswordFor(null);
    }
  };

  const togglePasswordVisibility = (username: string) => {
    const next = new Set(visiblePasswords);
    if (next.has(username)) next.delete(username);
    else next.add(username);
    setVisiblePasswords(next);
  };

  const exportToGoogleFormJSON = () => {
    const data = filteredQuestions.map(q => ({
      type: q.type === 'MCQ' ? 'MULTIPLE_CHOICE' : 'PARAGRAPH_TEXT',
      question: q.question,
      options: q.options,
      correctAnswer: q.correctAnswer,
      explanation: q.explanation,
      imageDescription: q.imageDescription
    }));
    const blob = new Blob([JSON.stringify({ title: "QuizGenius Export", items: data }, null, 2)], { type: 'application/json' });
    const url = URL.createObjectURL(blob);
    const link = document.createElement('a');
    link.href = url;
    link.download = "google_form_payload.json";
    link.click();
  };

  const downloadPerformanceCSV = (data: QuizAttempt[], filename: string) => {
    if (data.length === 0) return alert("No data available to export.");
    
    const headers = ["Username", "Module/Set", "Score", "Total", "Percentage", "Date", "Format"];
    const rows = data.map(att => [
      att.username,
      att.isCustomSet ? (quizSets.find(s => s.id === att.topicId)?.name || att.topicId) : att.topicId,
      att.score,
      att.total,
      `${Math.round((att.score / att.total) * 100)}%`,
      new Date(att.date).toLocaleString(),
      att.isCustomSet ? "Official" : "Practice"
    ]);

    const csvContent = [
      headers.join(","),
      ...rows.map(r => r.map(field => `"${field}"`).join(","))
    ].join("\n");

    const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });
    const url = URL.createObjectURL(blob);
    const link = document.createElement('a');
    link.href = url;
    link.download = filename;
    link.click();
  };

  const downloadDetailedAttemptReport = (attempt: QuizAttempt) => {
    if (!attempt.answers || attempt.answers.length === 0) return alert("No detailed question data available for this session.");
    
    const headers = ["Question Text", "Topic", "Student Answer", "Correct Answer", "Result"];
    const rows = attempt.answers.map(ans => {
      const q = questions.find(qu => qu.id === ans.questionId);
      return [
        q?.question || "Unknown Question",
        q?.topic || "Unknown Topic",
        ans.selected || "No Selection",
        q?.correctAnswer || "Unknown Key",
        ans.isCorrect ? "Correct" : "Incorrect"
      ];
    });

    const csvContent = [
      [`Report for ${attempt.username} - ${attempt.topicId} on ${new Date(attempt.date).toLocaleString()}`],
      [`Score: ${attempt.score}/${attempt.total} (${Math.round((attempt.score/attempt.total)*100)}%)`],
      [""],
      headers.join(","),
      ...rows.map(r => r.map(field => `"${field.toString().replace(/"/g, '""')}"`).join(","))
    ].join("\n");

    const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });
    const url = URL.createObjectURL(blob);
    const link = document.createElement('a');
    link.href = url;
    link.download = `${attempt.username}_${attempt.topicId}_detailed_report.csv`;
    link.click();
  };

  const studentPerformance = useMemo(() => {
    if (!viewingStudentRecords) return [];
    return attempts.filter(a => a.username === viewingStudentRecords).sort((a, b) => new Date(b.date).getTime() - new Date(a.date).getTime());
  }, [attempts, viewingStudentRecords]);

  const selectedAttemptDetail = useMemo(() => {
    if (!viewingAttemptDetailId) return null;
    return attempts.find(a => a.id === viewingAttemptDetailId);
  }, [attempts, viewingAttemptDetailId]);

  const quickAdd = (type: 'year' | 'subject' | 'topic') => {
    const val = prompt(`Enter new ${type} name:`);
    if (!val) return;
    if (type === 'year') onAddYear(val);
    else if (type === 'subject') onAddSubject(val);
    else if (type === 'topic') onAddTopic(val);
  };

  return (
    <div className="space-y-8 animate-fadeIn max-w-7xl mx-auto px-4 pb-20">
      {/* Universal Confirmation Modal */}
      {confirmDelete && (
        <div className="fixed inset-0 z-[200] bg-slate-900/60 backdrop-blur-md flex items-center justify-center p-6">
          <div className="bg-white w-full max-w-md rounded-[3rem] shadow-2xl p-10 animate-scaleUp border border-slate-100 text-center">
            <div className="w-20 h-20 bg-red-100 text-red-600 rounded-[1.5rem] flex items-center justify-center text-3xl mx-auto mb-6">⚠️</div>
            <h3 className="text-2xl font-black text-slate-900 mb-2">Delete Confirmation</h3>
            <p className="text-slate-500 font-medium mb-8">
              Are you sure you want to remove this {confirmDelete.type}? 
              {confirmDelete.name && <span className="block font-black text-slate-800 mt-2">"{confirmDelete.name}"</span>}
              This action can be undone for a short time after.
            </p>
            <div className="flex gap-4">
              <button onClick={() => setConfirmDelete(null)} className="flex-1 py-4 bg-slate-100 text-slate-500 rounded-2xl font-black text-[10px] uppercase tracking-widest hover:bg-slate-200 transition-all">Cancel</button>
              <button onClick={handleConfirmAction} className="flex-1 py-4 bg-red-600 text-white rounded-2xl font-black text-[10px] uppercase tracking-widest shadow-xl active:scale-95 transition-all">Confirm Delete</button>
            </div>
          </div>
        </div>
      )}

      {/* Password Reset Modal */}
      {editingPasswordFor && (
        <div className="fixed inset-0 z-[200] bg-slate-900/60 backdrop-blur-md flex items-center justify-center p-6">
          <div className="bg-white w-full max-w-md rounded-[3rem] shadow-2xl p-10 animate-scaleUp border border-slate-100">
            <div className="text-center mb-8">
              <div className="w-16 h-16 bg-indigo-100 text-indigo-600 rounded-2xl flex items-center justify-center text-2xl mx-auto mb-4">🔑</div>
              <h3 className="text-2xl font-black text-slate-900">Rotate Access Key</h3>
              <p className="text-slate-400 text-xs font-bold uppercase tracking-widest mt-1">User: {editingPasswordFor.username}</p>
            </div>
            <form onSubmit={handlePasswordResetSubmit} className="space-y-6">
              <div className="space-y-1">
                <label className="text-[10px] font-black uppercase text-slate-400 ml-2 tracking-widest">New Security Password</label>
                <input 
                  type="text"
                  autoFocus
                  value={editingPasswordFor.newPass} 
                  onChange={e => setEditingPasswordFor({...editingPasswordFor, newPass: e.target.value})} 
                  placeholder="Enter new password" 
                  className="w-full p-4 rounded-2xl border bg-slate-50 font-bold text-sm outline-none focus:ring-2 focus:ring-indigo-500/20"
                />
              </div>
              <div className="flex gap-3">
                <button type="button" onClick={() => setEditingPasswordFor(null)} className="flex-1 py-4 bg-slate-100 text-slate-500 rounded-2xl font-black text-[10px] uppercase tracking-widest">Cancel</button>
                <button type="submit" className="flex-1 py-4 bg-indigo-600 text-white rounded-2xl font-black text-[10px] uppercase tracking-widest shadow-lg">Save Update</button>
              </div>
            </form>
          </div>
        </div>
      )}

      {/* Improved Bank Edit Modal */}
      {editingQuestion && (
        <div className="fixed inset-0 z-[100] bg-slate-900/60 backdrop-blur-md flex items-center justify-center p-6">
          <div className="bg-white w-full max-w-5xl rounded-[3rem] shadow-2xl p-12 flex flex-col max-h-[90vh] overflow-y-auto animate-scaleUp">
            <div className="flex justify-between items-center mb-8">
              <div>
                <h3 className="text-3xl font-black text-slate-900">Refine Question</h3>
                <p className="text-slate-400 text-sm font-medium">Modify content, options, and metadata</p>
              </div>
              <button onClick={() => setEditingQuestion(null)} className="w-14 h-14 rounded-full bg-slate-100 flex items-center justify-center text-slate-400 hover:bg-slate-900 hover:text-white transition-all">✕</button>
            </div>
            
            <form onSubmit={handleEditSubmit} className="space-y-8">
              <div className="flex justify-between items-center bg-slate-50 p-4 rounded-2xl">
                <span className="text-[10px] font-black uppercase text-slate-400 tracking-widest">Math Rendering Preview</span>

              {showEditPreview && (

                      value={editingQuestion.correctAnswer}

                 <div className="space-y-2">

                    <label className="text-[10px] font-black uppercase text-slate-400 tracking-widest">Subject</label>

                    <label className="text-[10px] font-black uppercase text-slate-400 tracking-widest">Topic Allocation</label>

            <div className="flex justify-between items-center mb-8">
              <div className="flex items-center gap-4">
                {viewingAttemptDetailId && (

                  {viewingAttemptDetailId ? `${viewingStudentRecords}'s Detailed Results` : `${viewingStudentRecords}'s Performance Ledger`}

              <div className="flex items-center gap-4">

                          ✕
                        </button>
                      </div>
                    </div>
                  </div>

                  {selectedAttemptDetail?.answers ? (
                    selectedAttemptDetail.answers.map((ans, idx) => {
                      const q = questions.find(qu => qu.id === ans.questionId);
                      if (!q) return null;
                      return (
                        <div key={idx} className={`p-8 rounded-[2.5rem] border ${ans.isCorrect ? 'bg-green-50 border-green-100' : 'bg-red-50 border-red-100'}`}>

                             <span className="text-[10px] font-black uppercase bg-white px-3 py-1 rounded-lg shadow-sm border border-slate-100">Question {idx + 1}</span>

                             </div>

                              <h4 className="text-[8px] font-black uppercase text-slate-400 mb-1 tracking-widest">Marking Scheme / Explanation</h4>
                              <MathText text={q.explanation} className="text-xs text-slate-600 italic" />
                            </div>
                          )}

                      );

                    <div className="text-center py-20 text-slate-400">
                      <p>Detailed question history not available for this legacy attempt.</p>
                    </div>

            </div>
          </div>

                  <button onClick={() => handleProcessUpload(false)} disabled={isProcessing || !selectedFile} className="w-full py-5 bg-indigo-600 text-white rounded-[2rem] font-black text-lg shadow-2xl disabled:opacity-50 relative overflow-hidden transition-all active:scale-95">

            {showManualPreview && (
              <div className="p-8 bg-indigo-50/30 rounded-3xl border border-indigo-100 animate-fadeIn">
                 <h4 className="text-[9px] font-black uppercase text-indigo-400 mb-4 tracking-widest">Live Preview (KaTeX Enabled)</h4>

                  <div className="flex justify-between px-2">
                    <label className="text-[10px] font-black uppercase text-slate-400 tracking-widest">Year / Level</label>

                    />
                  </div>
                  <button type="submit" className="w-full py-4 bg-indigo-600 text-white rounded-2xl font-black text-xs uppercase tracking-widest shadow-xl shadow-indigo-100 hover:bg-slate-900 transition-all">Add to Roster</button>
               </form>
            </div>

            <div className="grid grid-cols-1 gap-4">
              {students.map(s => {
                const isPassVisible = visiblePasswords.has(s.username);
                return (
                  <div key={s.username} className="bg-white p-8 border border-slate-100 rounded-[2.5rem] shadow-sm flex flex-col md:flex-row items-center justify-between group hover:border-indigo-200 transition-all gap-6">
                    <div className="flex items-center gap-6 flex-1 min-w-0">
                      <div className="w-14 h-14 bg-indigo-600 rounded-2xl flex items-center justify-center text-white text-2xl font-black">{s.username.charAt(0).toUpperCase()}</div>
                      <div className="min-w-0">
                        <h3 className="text-xl font-black text-slate-900 truncate">{s.username}</h3>
                        <p className={`text-[10px] font-black uppercase tracking-widest ${s.isActive ? 'text-green-500' : 'text-amber-500'}`}>{s.isActive ? 'Verified Member' : 'Access Restricted / Pending'}</p>
                      </div>
                    </div>

                    <div className="flex items-center gap-4 bg-slate-50 px-6 py-3 rounded-2xl border border-slate-100">
                      <div className="flex flex-col">
                        <span className="text-[8px] font-black text-slate-400 uppercase tracking-widest mb-1">Student Passport & Security</span>
                        <span className="text-xs font-mono font-bold text-slate-600">
                          {isPassVisible ? s.password : '••••••••'}
                        </span>
                      </div>
                      <button 
                        onClick={() => togglePasswordVisibility(s.username)}
                        className="w-8 h-8 flex items-center justify-center bg-white text-slate-400 rounded-lg hover:text-indigo-600 transition-colors shadow-sm"
                        title={isPassVisible ? "Hide Secret" : "Audit Secret"}
                      >
                        {isPassVisible ? '🔒' : '👁️'}
                      </button>
                      <button 
                        onClick={() => setEditingPasswordFor({ username: s.username, newPass: '' })}
                        className="w-8 h-8 flex items-center justify-center bg-white text-slate-400 rounded-lg hover:text-indigo-600 transition-colors shadow-sm"
                        title="Force Password Override"
                      >
                        🔑
                      </button>
                    </div>

                    <div className="flex gap-3">
                      <button onClick={() => setViewingStudentRecords(s.username)} className="px-6 py-3 bg-slate-100 text-slate-600 rounded-xl font-black text-[10px] uppercase tracking-widest hover:bg-slate-900 hover:text-white transition-all">Performance</button>
                      <button onClick={() => onToggleUserStatus(s.username)} className={`px-6 py-3 rounded-xl font-black text-[10px] uppercase tracking-widest transition-all ${s.isActive ? 'bg-red-50 text-red-500' : 'bg-green-600 text-white'}`}>{s.isActive ? 'Revoke Access' : 'Grant Access'}</button>
                      <button onClick={() => setConfirmDelete({ type: 'student', id: s.username, name: s.username })} className="w-12 h-12 flex items-center justify-center bg-red-50 text-red-500 rounded-xl hover:bg-red-500 hover:text-white transition-all">✕</button>
                    </div>
                  </div>
                );
              })}
            </div>
          </div>
        )}

        {activeTab === 'manage' && (
          <div className="grid grid-cols-1 md:grid-cols-3 gap-8 animate-fadeIn">
             {[
               {label: 'Curriculums', items: years, onAdd: onAddYear}, 
               {label: 'Subjects', items: subjects, onAdd: onAddSubject}, 
               {label: 'Topics', items: topics, onAdd: onAddTopic}
             ].map((col, i) => (
                <div key={i} className="bg-white p-8 rounded-[3rem] border shadow-sm space-y-6">
                   <h2 className="text-xl font-black text-slate-900 uppercase tracking-tighter tracking-widest">{col.label} Ledger</h2>
                   <div className="flex gap-2">
                     <input 
                       id={`in-${i}`} 
                       placeholder={`New ${col.label.slice(0, -1)}`} 
                       className="flex-1 p-4 bg-slate-50 border rounded-2xl outline-none font-bold text-xs focus:ring-2 focus:ring-indigo-500/20" 
                     />
                     <button 
                       onClick={() => { const val = (document.getElementById(`in-${i}`) as HTMLInputElement).value; if(val) col.onAdd(val); (document.getElementById(`in-${i}`) as HTMLInputElement).value = ''; }} 
                       className="bg-indigo-600 text-white px-4 rounded-2xl font-black uppercase text-[10px] tracking-widest hover:bg-slate-900 transition-all"
                     >
                       Add
                     </button>
                   </div>
                   <div className="space-y-2 max-h-[50vh] overflow-y-auto pr-2 custom-scrollbar p-1">
                     {col.items.map(it => <div key={it.id} className="p-4 bg-slate-50 rounded-2xl font-black text-slate-700 text-sm border border-slate-100 hover:bg-indigo-50 hover:border-indigo-200 transition-all cursor-default">{it.name}</div>)}
                   </div>
                </div>
             ))}
          </div>
        )}
      </div>

      {/* Generalized Undo Toast */}
      {undoToast.show && (
        <div className="fixed bottom-10 right-10 z-[100] bg-slate-900 text-white px-8 py-5 rounded-[2.5rem] shadow-2xl flex items-center gap-8 border border-white/10 animate-slideUp">
           <div className="flex items-center gap-4">
              <div className="w-10 h-10 bg-red-500/20 text-red-400 rounded-xl flex items-center justify-center text-xl">🗑️</div>
              <div>
                 <p className="font-bold text-sm capitalize">{undoToast.type} removed</p>
                 <p className="text-[10px] text-slate-400 uppercase tracking-widest font-black">Restorable for 6s</p>
              </div>
           </div>
           <button onClick={handleUndo} className="bg-indigo-600 hover:bg-indigo-500 text-white px-8 py-3 rounded-2xl font-black text-xs transition-all flex items-center gap-2 group">
             <span className="group-hover:-translate-x-1 transition-transform">↩️</span> RESTORE
           </button>
        </div>
      )}

      <style>{`
        .custom-scrollbar::-webkit-scrollbar { width: 6px; }
        .custom-scrollbar::-webkit-scrollbar-thumb { background: #cbd5e1; border-radius: 10px; }
        .animate-fadeIn { animation: fadeIn 0.4s ease-out; }
        @keyframes fadeIn { from { opacity: 0; transform: translateY(10px); } to { opacity: 1; transform: translateY(0); } }
        @keyframes scaleUp { from { transform: scale(0.95); opacity: 0; } to { transform: scale(1); opacity: 1; } }
        .animate-scaleUp { animation: scaleUp 0.3s cubic-bezier(0.16, 1, 0.3, 1) forwards; }
        @keyframes slideUp { from { transform: translateY(100%); opacity: 0; } to { transform: translateY(0); opacity: 1; } }
        .animate-slideUp { animation: slideUp 0.4s cubic-bezier(0.16, 1, 0.3, 1) forwards; }
      `}</style>
    </div>
  );
};

export default AdminPanel;
