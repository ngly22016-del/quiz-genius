
import React, { useState, useMemo } from 'react';
import { Question, QuizAttempt, User, QuizSet, Year, Subject, Topic, UserAnswer, QuestionType } from '../types';
import Quiz from './Quiz';

interface UserPanelProps {
  user: User;
  questions: Question[];
  years: Year[];
  subjects: Subject[];
  topics: Topic[];
  quizSets: QuizSet[];
  attempts: QuizAttempt[];
  onRecordAttempt: (attempt: QuizAttempt) => void;
}

const UserPanel: React.FC<UserPanelProps> = ({ user, questions, years, subjects, topics, quizSets, attempts, onRecordAttempt }) => {
  const [selectedTopic, setSelectedTopic] = useState<string | null>(null);
  const [viewMode, setViewMode] = useState<'list' | 'quiz'>('list');
  const [searchTerm, setSearchTerm] = useState('');
  const [activeYear, setActiveYear] = useState<string>('All');
  const [activeSubject, setActiveSubject] = useState<string>('All');
  const [isTakingCustomSet, setIsTakingCustomSet] = useState(false);
  
  // Quiz Configuration State
  const [showQuizConfig, setShowQuizConfig] = useState<string | null>(null); // Topic name or Set ID
  const [quizLimit, setQuizLimit] = useState<number | 'all'>('all');
  const [quizFormat, setQuizFormat] = useState<QuestionType | 'MIXED'>('MIXED');

  const availableTopicNames: string[] = useMemo(() => Array.from(new Set(questions.map(q => q.topic))), [questions]);
  
  const userAttempts = attempts.filter(a => a.username === user.username);

  const assignedSets = useMemo(() => {
    return quizSets.filter(set => (set.assignedUsernames || []).includes(user.username));
  }, [quizSets, user.username]);

  const filteredTopics = useMemo(() => {
    return availableTopicNames.filter(topic => {
      const q = questions.find(qu => qu.topic === topic);
      const matchesSearch = topic.toLowerCase().includes(searchTerm.toLowerCase());
      const matchesYear = activeYear === 'All' || q?.year === activeYear;
      const matchesSubject = activeSubject === 'All' || q?.subject === activeSubject;
      return matchesSearch && matchesYear && matchesSubject;
    });
  }, [availableTopicNames, questions, searchTerm, activeYear, activeSubject]);

  const averageScore = userAttempts.length > 0 
    ? Math.round((userAttempts.reduce((acc, curr) => acc + (curr.score / curr.total), 0) / userAttempts.length) * 100) 
    : 0;

  const handleFinishQuiz = (score: number, total: number, answers: UserAnswer[]) => {
    if (selectedTopic) {
      onRecordAttempt({
        id: Date.now().toString(),
        username: user.username,
        topicId: selectedTopic,
        score,
        total,
        date: new Date().toISOString(),
        isCustomSet: isTakingCustomSet,
        answers: answers
      });
    }
    setViewMode('list');
    setSelectedTopic(null);
    setIsTakingCustomSet(false);
    setQuizLimit('all');
    setQuizFormat('MIXED');
  };

  const initiateQuizConfig = (topic: string, isCustom = false) => {
    setShowQuizConfig(topic);
    setIsTakingCustomSet(isCustom);
  };

  const handleStartQuiz = () => {
    if (!showQuizConfig) return;
    setSelectedTopic(showQuizConfig);
    setViewMode('quiz');
    setShowQuizConfig(null);
  };

  if (viewMode === 'quiz' && selectedTopic) {
    let baseQuestions: Question[] = [];
    if (isTakingCustomSet) {
      const set = quizSets.find(s => s.id === selectedTopic);
      baseQuestions = questions.filter(q => set?.questionIds.includes(q.id));
    } else {
      baseQuestions = questions.filter(q => q.topic === selectedTopic);
    }

    let finalQuestions = baseQuestions;
    if (quizFormat !== 'MIXED') {
      finalQuestions = finalQuestions.filter(q => q.type === quizFormat);
    }

    // Shuffle
    for (let i = finalQuestions.length - 1; i > 0; i--) {
      const j = Math.floor(Math.random() * (i + 1));
      [finalQuestions[i], finalQuestions[j]] = [finalQuestions[j], finalQuestions[i]];
    }

    if (quizLimit !== 'all') {
      finalQuestions = finalQuestions.slice(0, quizLimit as number);
    }

    if (finalQuestions.length === 0) {
      alert(`No questions found matching the "${quizFormat}" format. Starting with available questions.`);
      finalQuestions = baseQuestions.slice(0, quizLimit === 'all' ? baseQuestions.length : quizLimit as number);
    }

    return (
      <Quiz 
        topic={isTakingCustomSet ? quizSets.find(s => s.id === selectedTopic)?.name || 'Custom Quiz' : selectedTopic} 
        questions={finalQuestions} 
        username={user.username}
        onFinish={handleFinishQuiz} 
      />
    );
  }

  return (
    <div className="space-y-12 animate-fadeIn pb-24 max-w-7xl mx-auto px-4 md:px-0">
      {/* Quiz Configuration Modal */}
      {showQuizConfig && (
        <div className="fixed inset-0 z-[100] bg-slate-900/60 backdrop-blur-md flex items-center justify-center p-6">
          <div className="bg-white w-full max-w-xl rounded-[3.5rem] shadow-2xl p-12 animate-scaleUp border border-slate-100">
            <div className="text-center mb-10">
              <div className="w-20 h-20 bg-indigo-600 rounded-[2rem] flex items-center justify-center text-3xl mx-auto mb-4 shadow-xl shadow-indigo-100 text-white">⚡</div>
              <h3 className="text-3xl font-black text-slate-900 tracking-tight">Configure Session</h3>
              <p className="text-slate-500 font-medium">Topic: <span className="text-indigo-600 font-bold">{isTakingCustomSet ? quizSets.find(s => s.id === showQuizConfig)?.name : showQuizConfig}</span></p>
            </div>

            <div className="space-y-8">
              <div>
                <label className="text-[10px] font-black uppercase text-slate-400 mb-4 block tracking-[0.2em] ml-2">Question Format</label>
                <div className="grid grid-cols-2 md:grid-cols-4 gap-3">
                  {[
                    { id: 'MIXED', label: 'Mixed', icon: '🌀' },
                    { id: 'MCQ', label: 'MCQ', icon: '🔘' },
                    { id: 'STRUCTURE', label: 'Structured', icon: '📝' },
                    { id: 'ESSAY', label: 'Essay', icon: '✒️' }
                  ].map(f => (
                    <button
                      key={f.id}
                      onClick={() => setQuizFormat(f.id as any)}
                      className={`p-4 rounded-2xl border-2 transition-all flex flex-col items-center gap-2 ${
                        quizFormat === f.id ? 'border-indigo-600 bg-indigo-50 text-indigo-700 shadow-md' : 'border-slate-50 bg-slate-50 hover:border-slate-200 text-slate-500'
                      }`}
                    >
                      <span className="text-xl">{f.icon}</span>
                      <span className="text-[10px] font-black uppercase">{f.label}</span>
                    </button>
                  ))}
                </div>
              </div>

              <div>
                <label className="text-[10px] font-black uppercase text-slate-400 mb-4 block tracking-[0.2em] ml-2">Question Count</label>
                <div className="grid grid-cols-2 md:grid-cols-4 gap-3">
                  {[
                    { id: 5, label: '5 Qs' },
                    { id: 10, label: '10 Qs' },
                    { id: 20, label: '20 Qs' },
                    { id: 'all', label: 'All' }
                  ].map(q => (
                    <button
                      key={q.id}
                      onClick={() => setQuizLimit(q.id as any)}
                      className={`p-4 rounded-2xl border-2 transition-all text-[10px] font-black uppercase tracking-widest ${
                        quizLimit === q.id ? 'border-indigo-600 bg-indigo-50 text-indigo-700 shadow-md' : 'border-slate-50 bg-slate-50 hover:border-slate-200 text-slate-500'
                      }`}
                    >
                      {q.label}
                    </button>
                  ))}
                </div>
              </div>
            </div>

            <div className="flex gap-4 mt-12">
              <button 
                onClick={() => setShowQuizConfig(null)} 
                className="flex-1 py-5 bg-slate-100 text-slate-500 rounded-3xl font-black text-xs uppercase tracking-widest hover:bg-slate-200 transition-all"
              >
                Cancel
              </button>
              <button 
                onClick={handleStartQuiz} 
                className="flex-1 py-5 bg-indigo-600 text-white rounded-3xl font-black text-xs uppercase tracking-widest shadow-2xl shadow-indigo-100 active:scale-95 transition-all hover:bg-slate-900"
              >
                Start Quiz
              </button>
            </div>
          </div>
        </div>
      )}

      {/* Hero Section */}
      <div className="relative overflow-hidden bg-slate-900 rounded-[3.5rem] p-12 md:p-16 text-white shadow-2xl">
        <div className="absolute top-0 right-0 w-1/2 h-full bg-gradient-to-l from-indigo-500/20 to-transparent pointer-events-none"></div>
        <div className="relative z-10 flex flex-col md:flex-row items-center justify-between gap-12">
          <div className="flex flex-col md:flex-row items-center gap-10">
            <div className="w-28 h-28 bg-gradient-to-br from-indigo-400 to-purple-600 rounded-[2.5rem] flex items-center justify-center text-5xl font-black shadow-2xl ring-8 ring-white/5">
              {user.username.charAt(0).toUpperCase()}
            </div>
            <div className="text-center md:text-left space-y-2">
              <h1 className="text-5xl md:text-6xl font-black tracking-tight tracking-tighter">Hi, {user.username}!</h1>
              <p className="text-slate-400 text-lg font-medium opacity-80">Practice makes perfect. Ready to learn?</p>
            </div>
          </div>
          <div className="flex gap-6">
            <div className="bg-white/5 backdrop-blur-xl rounded-[2rem] p-8 border border-white/10 text-center min-w-[140px]">
              <p className="text-[10px] font-black uppercase text-indigo-400 mb-2 tracking-[0.2em]">Mastery</p>
              <p className="text-4xl font-black tracking-tighter">{averageScore}%</p>
            </div>
            <div className="bg-white/5 backdrop-blur-xl rounded-[2rem] p-8 border border-white/10 text-center min-w-[140px]">
              <p className="text-[10px] font-black uppercase text-indigo-400 mb-2 tracking-[0.2em]">Sessions</p>
              <p className="text-4xl font-black tracking-tighter">{userAttempts.length}</p>
            </div>
          </div>
        </div>
      </div>

      {assignedSets.length > 0 && (
        <section className="space-y-6">
          <div className="flex items-center gap-4">
             <div className="w-10 h-10 bg-indigo-600 rounded-xl flex items-center justify-center text-white text-lg">🎯</div>
             <h2 className="text-3xl font-black text-slate-900 tracking-tight">Assigned Assessments</h2>
          </div>
          <div className="grid grid-cols-1 md:grid-cols-2 lg:grid-cols-3 gap-8">
            {assignedSets.map(set => (
              <div key={set.id} className="bg-white rounded-[3rem] p-8 border-2 border-indigo-100 shadow-xl shadow-indigo-50 flex flex-col hover:border-indigo-500 transition-all group">
                <div className="mb-6 flex-1">
                  <div className="flex justify-between items-start mb-4">
                    <span className="text-[10px] font-black uppercase bg-indigo-600 text-white px-3 py-1 rounded-lg">Official Task</span>
                    <span className="text-[10px] font-bold text-slate-400">{set.questionIds.length} Questions</span>
                  </div>
                  <h3 className="text-2xl font-black text-slate-900 group-hover:text-indigo-600 transition-colors">{set.name}</h3>
                </div>
                <button 
                  onClick={() => initiateQuizConfig(set.id, true)}
                  className="w-full py-4 bg-indigo-600 text-white rounded-2xl font-black text-sm hover:bg-indigo-700 transition-all shadow-lg shadow-indigo-100"
                >
                  Start Assessment
                </button>
              </div>
            ))}
          </div>
        </section>
      )}

      <div className="grid grid-cols-1 lg:grid-cols-12 gap-12">
        <div className="lg:col-span-8 space-y-10">
          <div className="flex items-center gap-4">
             <div className="w-10 h-10 bg-slate-900 rounded-xl flex items-center justify-center text-white text-lg">📚</div>
             <h2 className="text-3xl font-black text-slate-900 tracking-tight">Practice Modules</h2>
          </div>
          
          <div className="bg-white p-6 rounded-[3rem] shadow-sm border border-slate-200 flex flex-col md:flex-row gap-4 items-center">
            <div className="relative flex-1 w-full">
              <span className="absolute left-6 top-1/2 -translate-y-1/2 text-slate-300">🔍</span>
              <input 
                type="text" 
                placeholder="Search by topic..." 
                className="w-full pl-14 pr-6 py-4 bg-slate-50 rounded-[2rem] outline-none font-medium focus:ring-2 focus:ring-indigo-500/20 transition-all"
                value={searchTerm}
                onChange={(e) => setSearchTerm(e.target.value)}
              />
            </div>
          </div>

          <div className="grid grid-cols-1 md:grid-cols-2 gap-8">
            {filteredTopics.length > 0 ? (
              filteredTopics.map(topic => {
                const topicQuestions = questions.filter(q => q.topic === topic);
                const firstQ = topicQuestions[0];
                const topicAttempts = userAttempts.filter(a => a.topicId === topic);
                const bestScore = topicAttempts.length > 0 
                  ? Math.max(...topicAttempts.map(a => Math.round((a.score/a.total)*100))) 
                  : 0;

                return (
                  <div key={topic} className="group bg-white rounded-[3.5rem] p-10 border border-slate-200 hover:border-indigo-500 transition-all flex flex-col shadow-sm hover:shadow-2xl hover:shadow-indigo-500/5 hover:-translate-y-2">
                    <div className="mb-8">
                      <div className="flex flex-wrap gap-2 mb-4">
                         <span className="text-[10px] font-black uppercase bg-indigo-50 px-3 py-1 rounded-full text-indigo-600 tracking-tighter">{firstQ?.year}</span>
                         <span className="text-[10px] font-black uppercase bg-amber-50 px-3 py-1 rounded-full text-amber-600 tracking-tighter">{firstQ?.subject}</span>
                      </div>
                      <h3 className="text-3xl font-black text-slate-900 mb-3 tracking-tighter leading-none">{topic}</h3>
                      <div className="flex items-center gap-4">
                        <span className="text-[10px] font-black text-slate-400 uppercase tracking-widest">{topicQuestions.length} Questions</span>
                        {bestScore > 0 && (
                          <div className="flex items-center gap-2">
                             <div className="w-2 h-2 rounded-full bg-green-500"></div>
                             <span className="text-[10px] font-black text-green-600 uppercase tracking-widest">{bestScore}% Best</span>
                          </div>
                        )}
                      </div>
                    </div>
                    
                    <button onClick={() => initiateQuizConfig(topic)} className="w-full py-5 bg-indigo-600 text-white rounded-[2rem] font-black text-sm shadow-xl shadow-indigo-100 hover:bg-slate-900 transition-all">
                      Start Practice
                    </button>
                  </div>
                );
              })
            ) : (
              <div className="col-span-full py-24 text-center bg-white rounded-[3rem] border-4 border-dashed border-slate-100">
                 <p className="text-4xl mb-4">🪐</p>
                 <p className="text-xl font-black text-slate-900">No matching modules found</p>
              </div>
            )}
          </div>
        </div>

        <div className="lg:col-span-4 space-y-10">
           <div className="bg-white rounded-[3rem] border border-slate-200 p-10 shadow-sm h-fit sticky top-32">
              <h3 className="text-[10px] font-black text-slate-400 uppercase tracking-[0.3em] mb-8">Recent History</h3>
              <div className="space-y-8">
                 {userAttempts.length > 0 ? (
                   userAttempts.slice(0, 10).map(att => {
                     const isCustom = att.isCustomSet;
                     const setName = isCustom ? quizSets.find(s => s.id === att.topicId)?.name : att.topicId;
                     const percent = Math.round((att.score/att.total)*100);
                     
                     return (
                       <div key={att.id} className="flex gap-5 items-start">
                          <div className={`w-3 h-3 rounded-full mt-1.5 flex-shrink-0 ${percent >= 80 ? 'bg-green-500' : percent >= 50 ? 'bg-amber-500' : 'bg-red-500'}`}></div>
                          <div className="flex-1 min-w-0">
                             <p className="text-sm font-black text-slate-800 truncate leading-none mb-1">{setName}</p>
                             <p className="text-[10px] font-bold text-slate-400">{new Date(att.date).toLocaleDateString()}</p>
                          </div>
                          <span className={`text-[10px] font-black px-2 py-1 rounded-lg ${percent >= 80 ? 'bg-green-50 text-green-600' : 'bg-slate-50 text-slate-500'}`}>
                             {percent}%
                          </span>
                       </div>
                     );
                   })
                 ) : (
                   <div className="text-center py-10">
                      <p className="text-slate-300 italic text-xs font-medium">Take a quiz to see results.</p>
                   </div>
                 )}
              </div>
           </div>
        </div>
      </div>
    </div>
  );
};

export default UserPanel;
