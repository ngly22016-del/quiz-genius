
import React, { useState } from 'react';
import { Question, UserAnswer } from '../types';
import MathText from './MathText';

interface QuizProps {
  topic: string;
  questions: Question[];
  username: string;
  onFinish: (score: number, total: number, answers: UserAnswer[]) => void;
}

function shuffleArray<T>(array: T[]): T[] {
  const shuffled = [...array];
  for (let i = shuffled.length - 1; i > 0; i--) {
    const j = Math.floor(Math.random() * (i + 1));
    [shuffled[i], shuffled[j]] = [shuffled[j], shuffled[i]];
  }
  return shuffled;
}

const Quiz: React.FC<QuizProps> = ({ topic, questions, username, onFinish }) => {
  const [quizQueue, setQuizQueue] = useState<Question[]>(() => {
    const shuffledQs = shuffleArray(questions);
    return shuffledQs.map((q: Question) => {
      if (q.type === 'MCQ' && q.options) {
        const indices = q.options.map((_, i) => i);
        const shuffledIndices = shuffleArray(indices);
        return {
          ...q,
          options: shuffledIndices.map(i => q.options[i]),
          optionImages: q.optionImages ? shuffledIndices.map(i => q.optionImages![i]) : undefined
        };
      }
      return q;
    });
  });

  const [currentIndex, setCurrentIndex] = useState(0);
  const [score, setScore] = useState(0);
  const [selectedOption, setSelectedOption] = useState<string | null>(null);
  const [longAnswer, setLongAnswer] = useState('');
  const [showResult, setShowResult] = useState(false);
  const [quizComplete, setQuizComplete] = useState(false);
  const [answers, setAnswers] = useState<UserAnswer[]>([]);

  const currentQuestion = quizQueue[currentIndex];
  const answeredCount = answers.length;
  const totalUnique = questions.length;
  const progressPercent = (answeredCount / totalUnique) * 100;

  const handleNext = () => {
    if (currentQuestion.type === 'MCQ') {
      const isCorrect = selectedOption === currentQuestion.correctAnswer;
      if (isCorrect) setScore(s => s + 1);
      setAnswers(prev => [...prev, { questionId: currentQuestion.id, selected: selectedOption || '', isCorrect }]);
    } else {
      setScore(s => s + 1);
      setAnswers(prev => [...prev, { questionId: currentQuestion.id, selected: longAnswer, isCorrect: true }]);
    }

    if (currentIndex < quizQueue.length - 1) {
      setCurrentIndex(c => c + 1);
      setSelectedOption(null);
      setLongAnswer('');
      setShowResult(false);
    } else {
      setQuizComplete(true);
    }
  };

  const handleSkip = () => {
    setQuizQueue(prev => [...prev, currentQuestion]);
    setSelectedOption(null);
    setLongAnswer('');
    setShowResult(false);
    setCurrentIndex(c => c + 1);
  };

  if (quizComplete) {
    const percentage = Math.round((score / totalUnique) * 100);
    return (
      <div className="max-w-2xl mx-auto py-12 px-4 animate-scaleUp">
        <div className="bg-white p-10 md:p-16 rounded-[3.5rem] shadow-2xl shadow-indigo-200/50 border border-slate-100 text-center relative overflow-hidden">
          <div className="absolute top-0 left-0 w-full h-2 bg-gradient-to-r from-indigo-500 via-purple-500 to-pink-500"></div>
          <div className="mb-10">
            <div className={`w-32 h-32 rounded-full mx-auto flex items-center justify-center text-4xl font-black mb-6 ring-8 ${
              percentage >= 80 ? 'bg-green-100 text-green-600 ring-green-50' : 
              percentage >= 50 ? 'bg-orange-100 text-orange-600 ring-orange-50' : 'bg-red-100 text-red-600 ring-red-50'
            } animate-bounceSlow`}>
              {percentage}%
            </div>
            <h2 className="text-4xl font-black text-slate-900 mb-2">Practice Complete!</h2>
            <p className="text-slate-500 font-medium">Excellent work, {username}.</p>
          </div>
          <button onClick={() => onFinish(score, totalUnique, answers)} className="w-full py-6 bg-indigo-600 text-white rounded-[2rem] font-black text-xl shadow-xl shadow-indigo-200 hover:bg-indigo-700 transition-all active:scale-95">Finish & Return</button>
        </div>
      </div>
    );
  }

  if (!currentQuestion) return null;

  return (
    <div className="max-w-4xl mx-auto space-y-8 animate-fadeIn pb-24 px-4">
      <div className="space-y-4">
        <div className="flex justify-between items-end">
          <div className="flex flex-col gap-1">
            <h2 className="text-3xl font-black text-slate-900 tracking-tight">{topic}</h2>
            <span className="text-[10px] font-black uppercase text-indigo-400 tracking-[0.2em]">{currentQuestion.type} MODE</span>
          </div>
          <div className="bg-white px-4 py-2 rounded-2xl shadow-sm border border-slate-100 flex items-center gap-2">
            <span className="text-xs font-black text-slate-600 uppercase tracking-widest">Completed {answeredCount} / {totalUnique}</span>
          </div>
        </div>
        <div className="overflow-hidden h-3 mb-4 flex rounded-full bg-slate-200 shadow-inner">
          <div style={{ width: `${progressPercent}%` }} className="flex flex-col justify-center bg-indigo-600 transition-all duration-700 ease-out rounded-full"></div>
        </div>
      </div>

      <div className="bg-white p-8 md:p-14 rounded-[3.5rem] shadow-2xl border border-slate-100 relative">
        <MathText text={currentQuestion.question} className="text-2xl md:text-3xl font-bold text-slate-900 mb-8 leading-[1.3] tracking-tight" />

        {currentQuestion.image && (
          <div className="mb-10 flex justify-center">
            <img src={currentQuestion.image} alt="Context" className="max-h-[300px] rounded-3xl border shadow-md" />
          </div>
        )}

        {currentQuestion.type === 'MCQ' ? (
          <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
            {currentQuestion.options.map((opt, i) => {
              const isCorrect = opt === currentQuestion.correctAnswer;
              const isSelected = selectedOption === opt;
              const optImg = currentQuestion.optionImages?.[i];
              
              let buttonClass = "border-slate-100 hover:border-indigo-200";
              if (showResult) {
                if (isCorrect) buttonClass = "border-green-500 bg-green-50 ring-4 ring-green-100";
                else if (isSelected) buttonClass = "border-red-500 bg-red-50";
                else buttonClass = "opacity-40";
              } else if (isSelected) {
                buttonClass = "border-indigo-600 bg-indigo-50 ring-4 ring-indigo-50";
              }

              return (
                <button 
                  key={i} 
                  disabled={showResult}
                  onClick={() => setSelectedOption(opt)}
                  className={`w-full text-left p-6 rounded-[2rem] border-2 transition-all flex flex-col gap-4 ${buttonClass}`}
                >
                  <div className="flex items-center gap-4">
                    <span className={`w-10 h-10 rounded-xl flex items-center justify-center font-black ${isSelected ? 'bg-indigo-600 text-white' : 'bg-slate-100 text-slate-400'}`}>
                      {String.fromCharCode(65 + i)}
                    </span>
                    <MathText text={opt} className="font-bold flex-1" />
                  </div>
                  {optImg && <img src={optImg} className="h-32 w-full object-contain rounded-xl bg-white border border-slate-50" alt="Option" />}
                </button>
              );
            })}
          </div>
        ) : (
          <div className="space-y-6">
            <textarea 
              value={longAnswer} 
              onChange={e => setLongAnswer(e.target.value)} 
              disabled={showResult}
              className="w-full h-48 p-6 rounded-[2rem] border-2 bg-slate-50 outline-none focus:border-indigo-500 transition-all font-medium"
              placeholder={`Write your ${currentQuestion.type.toLowerCase()} answer here...`}
            />
            {showResult && (
              <div className="p-6 bg-indigo-50 rounded-[2rem] border border-indigo-100 animate-fadeIn">
                <h4 className="text-[10px] font-black uppercase text-indigo-400 mb-2">Expert Model Answer</h4>
                <MathText text={currentQuestion.correctAnswer} />
              </div>
            )}
          </div>
        )}

        {showResult && currentQuestion.explanation && (
          <div className="mt-10 p-6 bg-slate-50 rounded-[2rem] border border-slate-100 italic animate-fadeIn">
            <h4 className="text-[10px] font-black uppercase text-slate-400 mb-2">Marking Scheme / Explanation</h4>
            <MathText text={currentQuestion.explanation} />
          </div>
        )}

        <div className="mt-12 flex justify-end gap-4">
          {!showResult ? (
            <>
              <button onClick={handleSkip} className="px-8 py-4 bg-slate-100 text-slate-600 rounded-2xl font-bold hover:bg-slate-200 transition-colors">Skip</button>
              <button 
                onClick={() => setShowResult(true)} 
                disabled={currentQuestion.type === 'MCQ' ? !selectedOption : !longAnswer.trim()} 
                className="px-12 py-4 bg-indigo-600 text-white rounded-2xl font-bold disabled:opacity-50 shadow-lg shadow-indigo-100 active:scale-95 transition-all"
              >
                Check Answer
              </button>
            </>
          ) : (
            <button onClick={handleNext} className="px-12 py-4 bg-indigo-600 text-white rounded-2xl font-bold shadow-lg shadow-indigo-100 active:scale-95 transition-all">
              {currentIndex < quizQueue.length - 1 ? 'Next Question' : 'Finish Session'}
            </button>
          )}
        </div>
      </div>
    </div>
  );
};

export default Quiz;
