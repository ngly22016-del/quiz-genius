
import React from 'react';
import { Question } from '../types';
import MathText from './MathText';

interface TopicDetailsProps {
  topic: string;
  questions: Question[];
  onBack: () => void;
  onStartQuiz: () => void;
}

const TopicDetails: React.FC<TopicDetailsProps> = ({ topic, questions, onBack, onStartQuiz }) => {
  return (
    <div className="max-w-4xl mx-auto space-y-8 animate-fadeIn pb-20">
      <div className="flex justify-between items-center">
        <div>
          <button onClick={onBack} className="text-slate-500 hover:text-indigo-600 font-bold mb-2 flex items-center gap-1 transition-colors">
            ← Back to Modules
          </button>
          <h1 className="text-3xl font-black text-slate-900 tracking-tight">{topic}</h1>
        </div>
        <button onClick={onStartQuiz} className="bg-indigo-600 text-white px-8 py-4 rounded-2xl font-black text-sm uppercase tracking-widest shadow-xl shadow-indigo-100 hover:scale-105 transition-all active:scale-95">
          Practice Now
        </button>
      </div>

      <div className="space-y-6">
        {questions.map((q, index) => (
          <div key={q.id} className="bg-white rounded-[3rem] border border-slate-200 p-8 md:p-10 shadow-sm relative overflow-hidden">
            <div className="flex gap-6 items-start mb-6">
              <span className="w-10 h-10 bg-indigo-600 text-white rounded-2xl flex items-center justify-center font-black flex-shrink-0 shadow-lg shadow-indigo-100">{index + 1}</span>
              <div className="flex-1">
                <div className="flex items-center gap-2 mb-2">
                  <span className={`text-[8px] font-black uppercase px-2 py-1 rounded-lg ${
                    q.type === 'MCQ' ? 'bg-blue-100 text-blue-600' : 
                    q.type === 'STRUCTURE' ? 'bg-orange-100 text-orange-600' : 'bg-purple-100 text-purple-600'
                  }`}>
                    {q.type}
                  </span>
                  <span className="text-[8px] font-black uppercase bg-slate-100 text-slate-400 px-2 py-1 rounded-lg">Difficulty: {q.difficulty}</span>
                </div>
                <MathText text={q.question} className="text-xl md:text-2xl font-bold text-slate-900 leading-tight" />
              </div>
            </div>
            
            {q.image && <img src={q.image} className="max-h-64 rounded-3xl mb-8 border mx-auto shadow-sm" alt="Question diagram" />}
            
            {q.type === 'MCQ' ? (
              <div className="grid grid-cols-1 md:grid-cols-2 gap-4 mb-8">
                {q.options.map((opt, i) => (
                  <div key={i} className={`p-5 rounded-3xl border-2 flex flex-col gap-3 transition-all ${opt === q.correctAnswer ? 'bg-green-50 border-green-200 ring-4 ring-green-50' : 'bg-slate-50 border-slate-100'}`}>
                    <div className="flex justify-between items-center">
                      <span className="font-black text-[10px] text-slate-400">OPTION {String.fromCharCode(65 + i)}</span>
                      {opt === q.correctAnswer && <span className="text-green-600 font-black text-[8px] uppercase tracking-widest bg-white px-2 py-1 rounded-lg shadow-sm">Key Answer</span>}
                    </div>
                    <MathText text={opt} className="text-sm font-bold" />
                    {q.optionImages?.[i] && <img src={q.optionImages[i]} className="h-24 w-full object-contain rounded-xl bg-white border border-slate-100" alt="Opt" />}
                  </div>
                ))}
              </div>
            ) : (
              <div className="mb-8 p-8 bg-green-50/30 rounded-[2.5rem] border-2 border-green-100">
                <h4 className="text-[10px] font-black uppercase text-green-600 mb-3 tracking-[0.2em]">Ideal Model Answer</h4>
                <MathText text={q.correctAnswer} className="font-medium text-slate-800" />
              </div>
            )}

            {q.explanation && (
              <div className="p-6 bg-slate-50 rounded-[2.5rem] border border-slate-100 flex gap-4 items-start">
                <div className="text-xl">💡</div>
                <div className="flex-1">
                  <h4 className="text-[10px] font-black uppercase text-slate-400 mb-1 tracking-widest">Marking Guidance</h4>
                  <MathText text={q.explanation} className="text-slate-600 text-sm leading-relaxed font-medium" />
                </div>
              </div>
            )}
          </div>
        ))}
      </div>
    </div>
  );
};

export default TopicDetails;
